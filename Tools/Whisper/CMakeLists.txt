cmake_minimum_required(VERSION 3.20)
project(TingQiaoWhisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(TINGQIAO_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options("-ffile-prefix-map=${TINGQIAO_PROJECT_ROOT}=.")
    get_filename_component(TINGQIAO_VENDOR_ROOT "${TINGQIAO_PROJECT_ROOT}/build/vendor" REALPATH)
    add_compile_options("-ffile-prefix-map=${TINGQIAO_VENDOR_ROOT}=vendor")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory(../../build/vendor/whisper.cpp whisper)
add_executable(TingQiaoWhisper main.cpp)
target_include_directories(TingQiaoWhisper PRIVATE ../../build/vendor/whisper.cpp/examples)
target_link_libraries(TingQiaoWhisper PRIVATE whisper)
add_library(TingQiaoVAD SHARED vad.cpp)
target_link_libraries(TingQiaoVAD PRIVATE whisper)
set_target_properties(TingQiaoVAD PROPERTIES INSTALL_NAME_DIR "@rpath")

option(TINGQIAO_BUILD_CONVERTER_TEST "Build converter validation" OFF)
if(TINGQIAO_BUILD_CONVERTER_TEST)
    add_executable(KomixConversionTests ../../Tests/KomixConversionTests.cpp)
    target_compile_options(KomixConversionTests PRIVATE -UNDEBUG)
    target_include_directories(KomixConversionTests PRIVATE . ../../build/vendor/whisper.cpp/examples)
    target_link_libraries(KomixConversionTests PRIVATE whisper)
endif()
