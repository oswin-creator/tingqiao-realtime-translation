cmake_minimum_required(VERSION 3.20)
project(TingQiaoQwen LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(TINGQIAO_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options("-ffile-prefix-map=${TINGQIAO_PROJECT_ROOT}=.")
    get_filename_component(TINGQIAO_VENDOR_ROOT "${TINGQIAO_PROJECT_ROOT}/build/vendor" REALPATH)
    add_compile_options("-ffile-prefix-map=${TINGQIAO_VENDOR_ROOT}=vendor")
endif()
set(TRANSCRIBE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
add_subdirectory(../../build/vendor/transcribe.cpp transcribe)
target_include_directories(transcribe PUBLIC ../../build/vendor/transcribe.cpp/include)
add_executable(TingQiaoQwen main.cpp)
target_include_directories(TingQiaoQwen PRIVATE ../../build/vendor/whisper.cpp/examples)
target_link_libraries(TingQiaoQwen PRIVATE transcribe)
